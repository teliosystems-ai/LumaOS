cmake_minimum_required(VERSION 3.16)
project(luma_speech_engines LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  message(FATAL_ERROR "the initial speech profile is amd64 CPU only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_OPENVINO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_HIP OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
foreach(feature SSE42 AVX AVX2 AVX_VNNI AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16
                BMI2 FMA F16C AMX_TILE AMX_INT8 AMX_BF16)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(backend METAL MUSA OPENCL OPENVINO WEBGPU ZDNN VIRTGPU VIRTGPU_BACKEND ET HEXAGON ZENDNN)
  set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper-build EXCLUDE_FROM_ALL)
add_library(luma_asr_engine STATIC asr_engine.cpp)
target_compile_options(luma_asr_engine PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
target_link_libraries(luma_asr_engine PUBLIC whisper)
target_include_directories(luma_asr_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(luma-asr-worker asr_worker.cpp)
target_compile_options(luma-asr-worker PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
target_link_options(luma-asr-worker PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack)
target_link_libraries(luma-asr-worker PRIVATE luma_asr_engine)
